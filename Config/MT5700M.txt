# MT5700M 独立插件配置层（方案 A）
# 由原 Config/GENERAL.txt 中的 MT5700M 段整理而来，仅在 MT_MODE=MT5700M 时叠加。
# 依赖：luci-app-mt5700m + sms-tool_q + ubus-at-daemon（均来自 FUjr/QModem feed）
# 严格禁止与 Config/MT5700.txt / luci-app-mt5700 同时启用。

# LuCI 壳
CONFIG_PACKAGE_luci-app-mt5700m=y
CONFIG_PACKAGE_luci-i18n-mt5700m-zh-cn=y

# MT5700M 依赖组件（luci-app-mt5700m 的 LUCI_DEPENDS）
CONFIG_PACKAGE_ubus-at-daemon=y
CONFIG_PACKAGE_sms-tool_q=y

# 说明：luci-app-mt5700m 的 Rust AT 后端（/usr/bin/at-webserver，同时提供
# /usr/sbin/mt5700m-at 软链）与 www/5700 WebUI 前端不是独立软件包，而是在编译时
# 由 Scripts/Packages.sh 的 FOLD_MT5700M 交叉编译后“折叠”进 LuCI 壳，
# 因此这里无需（也不能）设置 CONFIG_PACKAGE_at-webserver=y。

# 显式排除 MT5700（互斥，防止 make defconfig 被其他配置意外拉入）
CONFIG_PACKAGE_luci-app-mt5700=n
CONFIG_PACKAGE_luci-i18n-mt5700-zh-cn=n
