# ===== Fibocom FM350-GL 模组：luci-app-fm350 =====
#
# 包来源：https://github.com/LianXia233/luci-app-fm350（分支 main，版本 1.0.2-r1）
# 构成：LuCI JS 界面 + Rust 后端守护 fm350d（/usr/sbin/fm350d）+ rpcd ucode 插件。
#
# 四条打包特性决定了这里该写什么、不该写什么：
#   1. 上游 Makefile 只 include rules.mk / package.mk，不使用 feeds/luci/luci.mk，
#      因此不依赖 LuCI feed 的编译框架，源码树与 SDK 都能构建。
#   2. 菜单通过 /usr/share/luci/menu.d/*.json 注册，无 Lua 控制器、无 .ut 模板。
#   3. 界面文案为中文硬编码，不使用 .lmo 翻译文件 —— 因此不存在
#      luci-i18n-fm350-zh-cn 这个包，写了就是一条无人认领的无效配置。
#   4. 后端 fm350d 由 cargo 交叉编译产出（musl 静态链接），不是独立软件包，
#      不能写 CONFIG_PACKAGE_fm350d=y。
CONFIG_PACKAGE_luci-app-fm350=y

# Makefile DEPENDS 逐项显式选中。
# 理由：依赖若只靠 DEPENDS 隐式解析，上游一旦调整打包、或某个依赖在别的配置层
# 被置 =n，make defconfig 之后会静默缺包 —— 界面能装但功能不全，比直接编不过更难查。
# kmod-usb-net-rndis / kmod-usb-serial-option 在 Config/GENERAL.txt 中已是 =y，
# 这里重复声明只为自解释；后写覆盖先写，结果一致。

CONFIG_PACKAGE_rpcd=y
CONFIG_PACKAGE_rpcd-mod-ucode=y
