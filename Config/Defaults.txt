# ===== 显式配置文件：系统默认（局域网 / 无线）=====
# 本文件是「后台地址 / 后台密码 / 主机名 / Wi-Fi 及频宽国家码」等默认配置的
# 唯一来源，由 Scripts/Settings.sh 在编译期读取并写入固件。
# 优先级：工作流显式传入的 WRT_* 输入 > 本文件默认值 > 脚本内置兜底。
# 修改后无需改动任何工作流 / 脚本，重新触发编译即生效。
# 注意：本文件被脚本 source 解析，值请勿包含未转义的空格或 #（行内注释）。

# ---- 局域网 ----
LAN_IP=192.168.10.1          # 后台地址（路由器 LAN IP，Web 管理界面访问地址）
LAN_PASSWORD=无              # 后台密码（仅用于 Release 说明展示；OpenWrt 默认
                             # 首次开机无密码，「无 / 留空」即不设置密码）
HOST_NAME=OWRT               # 默认主机名（网络标识）

# ---- 无线（WiFi6 默认 · AP3000M 等 802.11ax 设备）----
# 无后缀变量为 WiFi6 世代默认值（保持历史兼容）；Settings.sh 按设备世代选择：
# AP3000M（WiFi6）取本节，H5000M（WiFi7）取下方 _WIFI7 后缀节。
WIFI_SSID=OWRT               # Wi-Fi 名称（2.4G / 5G 共用）
WIFI_PASSWORD=12345678       # Wi-Fi 密码
WIFI_ENCRYPTION=psk-mixed    # 加密方式：mixed WPA/WPA2 PSK (CCMP)（psk-mixed）
WIFI_COUNTRY=CN              # 国家/地区码
WIFI_2G_WIDTH=40             # 2.4G 频宽（MHz）
WIFI_5G_WIDTH=160            # 5G 频宽（MHz）

# ---- 无线（WiFi7 默认 · H5000M 等 802.11be 设备）----
# 仅当 Settings.sh 识别到 WiFi7 设备（H5000M）时启用。注意：320MHz 仅限 6GHz
# 频段（802.11be 特性），H5000M 为 2.4G/5G 双频设备，5G 频段上限同为 160MHz。
WIFI_SSID_WIFI7=OWRT             # Wi-Fi 名称（2.4G / 5G 共用）
WIFI_PASSWORD_WIFI7=12345678     # Wi-Fi 密码
WIFI_ENCRYPTION_WIFI7=psk-mixed  # 加密方式：mixed WPA/WPA2 PSK (CCMP)（psk-mixed）
WIFI_COUNTRY_WIFI7=CN            # 国家/地区码
WIFI_2G_WIDTH_WIFI7=40           # 2.4G 频宽（MHz）
WIFI_5G_WIDTH_WIFI7=160          # 5G 频宽（MHz，5G 频段 802.11be 上限同为 160MHz）
